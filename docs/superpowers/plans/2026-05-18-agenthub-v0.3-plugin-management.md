# agenthub v0.3 — Pillar 4 Plugin Management Implementation Plan

> **For agentic workers:** Use superpowers:subagent-driven-development to implement this plan task-by-task.

**Goal:** Build Pillar 4 (Plugin Management) on top of v0.2.1. Enable `agenthub plugin add/ls/show/rm/update` to manage Claude Code marketplace plugins and OpenCode npm plugins via per-tool **PluginRef** files. Profile `[plugins].include` triggers install/uninstall during `apply`, delegating to each tool's native install mechanism (no symlinks).

**Architecture:** Each plugin is a "reference" (not vendored) — `hub/plugins/<tool>/<name>.toml` holds the URL/version. `apply` reads profile's plugin list, fetches each PluginRef, and calls the target's `install_plugin()` method which delegates to `claude plugin install`, Codex's equivalent, or `bun install`. Last-apply manifest tracks plugins so removing from profile triggers `uninstall_plugin()`.

**Tech stack:** No new pip deps. Uses existing `run_git`-style `run_subprocess` pattern (extend `core/git_subprocess.py` → `core/subprocess_utils.py` or split).

**Reference spec:** `docs/superpowers/specs/2026-05-15-alba-agent-tool-hub-design.md` §5 (Plugin management).

**Scope of v0.3:**
- PluginRef schema: marketplace (Claude/Codex) + npm (OpenCode)
- CLI: `plugin add / ls / show / rm / update`
- Profile-driven install: `apply` reads `[plugins].include`, calls target adapter
- Manifest tracks installed plugins per target
- Claude Code: fully working (verified via mocked CLI in tests)
- Codex: best-effort impl; specifics still TBD in spec §1.6 — stubbed with clear error if invoked, full impl in v0.3.x
- OpenCode: npm via bun (or npm fallback) — modifies `~/.config/opencode/package.json`

**Out of scope (v0.3.x or later):**
- `agenthub plugin export --format claude-plugin-marketplace` (Phase 2)
- Per-skill granularity inside a plugin (deliberate: all-or-nothing)
- Plugin pre-install hooks
- Codex specifics deferred to v0.3.2 (v0.3.1 used for the post-v0.3.0 simplify cleanup tag, mirroring the v0.2.x pattern). Codex completion lands after `~/.codex/plugins/` install CLI is verified on a real machine.

**Opportunistic backlog fix bundled in this plan:**
- `_hub_diverges_from_lock` network-failure should NOT silently declare CONFLICT (v0.2 review #2). Convert to tri-state and let CLI distinguish "network unreachable" from "local edits".

```python
from __future__ import annotations

from dataclasses import dataclass, field
from enum import StrEnum
from pathlib import Path

import tomlkit

from agenthub import API_VERSION
from agenthub.core.envelope import EnvelopeError, parse_envelope


class PluginError(Exception):
    """Raised on PluginRef load/save errors."""


class PluginType(StrEnum):
    MARKETPLACE = "marketplace"   # Claude Code, Codex (git-hosted marketplaces)
    NPM = "npm"                   # OpenCode (npm-published plugins)


@dataclass(frozen=True, slots=True)
class PluginRef:
    type: PluginType
    # Marketplace fields (when type == MARKETPLACE)
    marketplace: str = ""
    plugin: str = ""
    # NPM fields (when type == NPM)
    npm_package: str = ""
    # Common
    version: str = "latest"
    description: str = ""
    tags: tuple[str, ...] = field(default_factory=tuple)


def _validate(ref: PluginRef) -> None:
    if ref.type is PluginType.MARKETPLACE:
        if not ref.marketplace or not ref.plugin:
            raise PluginError(
                "marketplace plugin requires both [plugin].marketplace and [plugin].plugin"
            )
    elif ref.type is PluginType.NPM:
        if not ref.npm_package:
            raise PluginError("npm plugin requires [plugin].npm_package")


def load_plugin_ref(path: Path) -> PluginRef:
    if not path.exists():
        raise PluginError(f"plugin ref not found: {path}")
    doc = tomlkit.parse(path.read_text())
    try:
        parse_envelope(doc, expected_kind="PluginRef")
    except EnvelopeError as e:
        raise PluginError(str(e)) from e
    plugin = doc.get("plugin", {})
    type_str = plugin.get("type", "")
    try:
        ptype = PluginType(type_str)
    except ValueError as e:
        raise PluginError(f"unknown plugin type: {type_str!r}") from e
    meta = doc.get("meta", {})
    ref = PluginRef(
        type=ptype,
        marketplace=plugin.get("marketplace", ""),
        plugin=plugin.get("plugin", ""),
        npm_package=plugin.get("npm_package", ""),
        version=plugin.get("version", "latest"),
        description=meta.get("description", ""),
        tags=tuple(meta.get("tags", []) or []),
    )
    _validate(ref)
    return ref


def save_plugin_ref(path: Path, ref: PluginRef) -> None:
    _validate(ref)
    doc = tomlkit.document()
    doc["apiVersion"] = API_VERSION
    doc["kind"] = "PluginRef"
    if ref.description or ref.tags:
        meta = tomlkit.table()
        if ref.description:
            meta["description"] = ref.description
        if ref.tags:
            meta["tags"] = list(ref.tags)
        doc["meta"] = meta
    plug = tomlkit.table()
    plug["type"] = ref.type.value
    if ref.type is PluginType.MARKETPLACE:
        plug["marketplace"] = ref.marketplace
        plug["plugin"] = ref.plugin
    elif ref.type is PluginType.NPM:
        plug["npm_package"] = ref.npm_package
    if ref.version and ref.version != "latest":
        plug["version"] = ref.version
    doc["plugin"] = plug
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(tomlkit.dumps(doc))
```

```python
from __future__ import annotations

import json
from pathlib import Path

from agenthub.core.git_subprocess import run_subprocess
from agenthub.core.plugin import PluginRef


class ClaudeTarget:
    name = "claude"

    def detect(self) -> bool:
        return self.global_root().exists()

    def global_root(self) -> Path:
        return Path.home() / ".claude"

    def project_root(self, project_path: Path) -> Path:
        return project_path / ".claude"

    def content_subdir(self, kind: str) -> str:
        return {"skill": "skills", "command": "commands", "agent": "agents"}[kind]

    def install_plugin(self, ref: PluginRef) -> None:
        # 1) Register marketplace (idempotent — claude CLI tolerates re-add)
        rc, _, err = run_subprocess(
            ["claude", "plugin", "marketplace", "add", ref.marketplace]
        )
        if rc != 0:
            raise RuntimeError(f"claude plugin marketplace add failed: {err.strip()}")
        # 2) Install plugin
        install_cmd = ["claude", "plugin", "install", ref.plugin]
        if ref.version and ref.version != "latest":
            install_cmd.extend(["--version", ref.version])
        rc, _, err = run_subprocess(install_cmd)
        if rc != 0:
            raise RuntimeError(f"claude plugin install failed: {err.strip()}")

    def uninstall_plugin(self, name: str) -> None:
        rc, _, err = run_subprocess(["claude", "plugin", "uninstall", name])
        if rc != 0 and "not installed" not in err.lower():
            raise RuntimeError(f"claude plugin uninstall failed: {err.strip()}")

    def list_installed_plugins(self) -> set[str]:
        manifest = self.global_root() / "plugins" / "installed_plugins.json"
        if not manifest.exists():
            return set()
        try:
            data = json.loads(manifest.read_text())
        except json.JSONDecodeError:
            return set()
        if isinstance(data, list):
            return {item.get("name", "") for item in data if isinstance(item, dict)}
        if isinstance(data, dict):
            return set(data.keys())
        return set()
```

```python
from __future__ import annotations

import json
import shutil
from pathlib import Path

from agenthub.core.git_subprocess import run_subprocess
from agenthub.core.plugin import PluginRef


def _package_json(global_root: Path) -> Path:
    return global_root / "package.json"


def _read_dependencies(pkg: Path) -> dict[str, str]:
    if not pkg.exists():
        return {}
    try:
        data = json.loads(pkg.read_text())
    except json.JSONDecodeError:
        return {}
    deps = data.get("dependencies")
    return dict(deps) if isinstance(deps, dict) else {}


def _write_package_json(pkg: Path, deps: dict[str, str]) -> None:
    if pkg.exists():
        data = json.loads(pkg.read_text())
    else:
        data = {"name": "opencode-deps", "private": True}
    data["dependencies"] = dict(sorted(deps.items()))
    pkg.write_text(json.dumps(data, indent=2) + "\n")


def _installer_cmd() -> list[str]:
    """Pick bun if available, else npm."""
    if shutil.which("bun"):
        return ["bun", "install"]
    if shutil.which("npm"):
        return ["npm", "install"]
    raise RuntimeError("neither bun nor npm found on PATH")


class OpenCodeTarget:
    name = "opencode"

    def detect(self) -> bool:
        return self.global_root().exists()

    def global_root(self) -> Path:
        return Path.home() / ".config" / "opencode"

    def project_root(self, project_path: Path) -> Path:
        return project_path / ".opencode"

    def content_subdir(self, kind: str) -> str:
        return {"skill": "skills", "command": "command", "agent": "agent"}[kind]

    def install_plugin(self, ref: PluginRef) -> None:
        pkg = _package_json(self.global_root())
        pkg.parent.mkdir(parents=True, exist_ok=True)
        deps = _read_dependencies(pkg)
        version = ref.version if ref.version else "latest"
        deps[ref.npm_package] = version
        _write_package_json(pkg, deps)
        rc, _, err = run_subprocess(_installer_cmd(), cwd=self.global_root())
        if rc != 0:
            raise RuntimeError(f"npm/bun install failed: {err.strip()}")

    def uninstall_plugin(self, name: str) -> None:
        pkg = _package_json(self.global_root())
        if not pkg.exists():
            return
        deps = _read_dependencies(pkg)
        if name not in deps:
            return
        deps.pop(name)
        _write_package_json(pkg, deps)
        rc, _, err = run_subprocess(_installer_cmd(), cwd=self.global_root())
        if rc != 0:
            raise RuntimeError(f"npm/bun install (after removal) failed: {err.strip()}")

    def list_installed_plugins(self) -> set[str]:
        return set(_read_dependencies(_package_json(self.global_root())).keys())
```

```python
from __future__ import annotations

from pathlib import Path

import click

from agenthub.cli._ctx import hub_path_from_ctx
from agenthub.core.plugin import (
    PluginError,
    PluginRef,
    PluginType,
    load_plugin_ref,
    save_plugin_ref,
)

_TARGETS = ("claude", "codex", "opencode")


@click.group("plugin")
def plugin_grp() -> None:
    """Manage plugin references in the hub."""


def _ref_path(hub: Path, target: str, plugin_name: str) -> Path:
    return hub / "plugins" / target / f"{plugin_name}.toml"


def _ref_name_from_npm(npm_package: str) -> str:
    """Derive a hub filename from an npm package: strip @scope, use last segment."""
    if "/" in npm_package:
        return npm_package.rsplit("/", 1)[-1]
    return npm_package.lstrip("@")


@plugin_grp.command("add")
@click.option("-t", "--target", "target", type=click.Choice(_TARGETS), required=True)
@click.option("--marketplace", "marketplace_url", default=None)
@click.option("--plugin", "plugin_name", default=None)
@click.option("--npm-package", "npm_package", default=None)
@click.option("--version", default="latest")
@click.option("--description", default="")
@click.option("--tags", default="", help="Comma-separated tag list")
@click.pass_context
def add_cmd(
    ctx: click.Context,
    target: str,
    marketplace_url: str | None,
    plugin_name: str | None,
    npm_package: str | None,
    version: str,
    description: str,
    tags: str,
) -> None:
    """Register a plugin reference for a target."""
    hub = hub_path_from_ctx(ctx)
    tag_tuple = tuple(t.strip() for t in tags.split(",") if t.strip())
    if npm_package:
        if target != "opencode":
            raise click.ClickException("--npm-package is only valid for -t opencode")
        ref = PluginRef(
            type=PluginType.NPM,
            npm_package=npm_package,
            version=version,
            description=description,
            tags=tag_tuple,
        )
        name = _ref_name_from_npm(npm_package)
    else:
        if not (marketplace_url and plugin_name):
            raise click.ClickException(
                "marketplace target requires --marketplace and --plugin"
            )
        ref = PluginRef(
            type=PluginType.MARKETPLACE,
            marketplace=marketplace_url,
            plugin=plugin_name,
            version=version,
            description=description,
            tags=tag_tuple,
        )
        name = plugin_name
    dest = _ref_path(hub, target, name)
    if dest.exists():
        raise click.ClickException(f"plugin ref already exists: {dest}")
    save_plugin_ref(dest, ref)
    click.echo(f"added: plugin ref at {dest}")
    click.echo("not in any profile/project yet; add to enable.")


@plugin_grp.command("ls")
@click.option("-t", "--target", "target", type=click.Choice(_TARGETS), default=None)
@click.pass_context
def ls_cmd(ctx: click.Context, target: str | None) -> None:
    """List plugin references in the hub, optionally filtered by target."""
    hub = hub_path_from_ctx(ctx)
    targets = [target] if target else list(_TARGETS)
    any_found = False
    for t in targets:
        base = hub / "plugins" / t
        if not base.exists():
            continue
        for ref_file in sorted(base.glob("*.toml")):
            try:
                ref = load_plugin_ref(ref_file)
            except PluginError as e:
                click.echo(f"WARN: {ref_file}: {e}")
                continue
            descriptor = (
                f"{ref.marketplace}/{ref.plugin}@{ref.version}"
                if ref.type is PluginType.MARKETPLACE
                else f"{ref.npm_package}@{ref.version}"
            )
            click.echo(f"{t:10s} {ref_file.stem:24s} {descriptor}")
            any_found = True
    if not any_found:
        click.echo("(no plugin refs)")


@plugin_grp.command("show")
@click.argument("name")
@click.option("-t", "--target", "target", type=click.Choice(_TARGETS), required=True)
@click.pass_context
def show_cmd(ctx: click.Context, name: str, target: str) -> None:
    """Print a plugin reference's TOML."""
    hub = hub_path_from_ctx(ctx)
    p = _ref_path(hub, target, name)
    if not p.exists():
        raise click.ClickException(f"no plugin ref at {p}")
    click.echo(p.read_text())


@plugin_grp.command("rm")
@click.option("-t", "--target", "target", type=click.Choice(_TARGETS), required=True)
@click.option("--plugin", "plugin_name", required=True)
@click.option("--yes", "-y", is_flag=True)
@click.pass_context
def rm_cmd(ctx: click.Context, target: str, plugin_name: str, yes: bool) -> None:
    """Remove a plugin reference (does NOT uninstall from the target tool)."""
    hub = hub_path_from_ctx(ctx)
    p = _ref_path(hub, target, plugin_name)
    if not p.exists():
        raise click.ClickException(f"no plugin ref at {p}")
    if not yes and not click.confirm(f"remove plugin ref {p}?", default=True):
        return
    p.unlink()
    click.echo(f"removed: {p}")
```

```python
def _plugin_plan(
    hub: Path, target: Target, includes: list[str], manifest: LastApply | None
) -> list[PlanItem]:
    plugin_root = hub / "plugins" / target.name
    if not plugin_root.exists():
        return []
    # Resolve includes (literal names + tags) against PluginRef metadata
    available = {}
    for f in plugin_root.glob("*.toml"):
        # If we want tag-based @include resolution, load_plugin_ref + match tag
        available[f.stem] = f
    desired = set()
    for entry in includes:
        if entry.startswith("@"):
            tag = entry[1:]
            for name, path in available.items():
                from agenthub.core.plugin import load_plugin_ref
                try:
                    ref = load_plugin_ref(path)
                    if tag in ref.tags:
                        desired.add(name)
                except Exception:
                    continue
        else:
            if entry in available:
                desired.add(entry)
    previously_managed = _previously_managed_plugins(manifest, target_name=target.name)
    items: list[PlanItem] = []
    for name in sorted(desired):
```

```python
    content_root: Path, intent: SourceIntent, lock: SourceLock, cache_root: Path,

    """Returns (diverges, network_error).

    diverges=True means user edited locally; network_error=non-None means we couldn't
    fetch the snapshot to compare (treat update conservatively: skip but don't claim conflict).
    """
    snapshot_cache = cache_path_for(intent.url + "#" + lock.sha, cache_root)
    if not (snapshot_cache / ".git").exists():
        snapshot_cache.parent.mkdir(parents=True, exist_ok=True)
        rc, _, err = run_git(["clone", "--no-tags", "--depth=1", intent.url, str(snapshot_cache)])
        if rc != 0:
            return False, f"clone failed: {err.strip()}"
        rc, _, err = run_git(["fetch", "--depth=1", "origin", lock.sha], cwd=snapshot_cache)
        if rc != 0:
            return False, f"fetch failed: {err.strip()}"
        rc, _, _ = run_git(["checkout", "FETCH_HEAD"], cwd=snapshot_cache)
        if rc != 0:
            return False, "checkout failed"
    snapshot_root = (snapshot_cache / intent.path) if intent.path else snapshot_cache
    if not snapshot_root.exists():
        return True, None
    return _trees_differ(content_root, snapshot_root), None
```

```python
def test_full_lifecycle_add_apply_remove(tmp_hub: Path, fake_home: Path):
    """add → apply (install) → remove from profile → apply (uninstall)."""
    runner = CliRunner()
    runner.invoke(main, ["init", "--hub", str(tmp_hub)])
    # Add plugin ref
    runner.invoke(main, [
        "plugin", "add", "-t", "claude",
        "--marketplace", "https://github.com/obra/superpowers",
        "--plugin", "superpowers", "--version", "5.0.7",
    ])
    # Create a profile that includes it
    p = tmp_hub / "profiles" / "claude" / "default"
    p.mkdir(parents=True)
    doc = tomlkit.document()
    doc["apiVersion"] = "agenthub.dev/v1"
    doc["kind"] = "Profile"
    tbl = tomlkit.table(); tbl["include"] = ["superpowers"]; doc["plugins"] = tbl
    (p / "profile.toml").write_text(tomlkit.dumps(doc))
    # Apply
    installs = []
    uninstalls = []
    def fake_install(self, ref): installs.append(ref.plugin)
    def fake_uninstall(self, name): uninstalls.append(name)
    with patch("agenthub.targets.claude.ClaudeTarget.install_plugin", autospec=True, side_effect=fake_install):
        with patch("agenthub.targets.claude.ClaudeTarget.uninstall_plugin", autospec=True, side_effect=fake_uninstall):
            r = runner.invoke(main, ["apply", "-t", "claude", "--profile", "default", "--yes"])
            assert r.exit_code == 0, r.output
            assert installs == ["superpowers"]
            # Now drop from profile
            doc["plugins"]["include"] = []  # type: ignore
            (p / "profile.toml").write_text(tomlkit.dumps(doc))
            r2 = runner.invoke(main, ["apply", "-t", "claude", "--profile", "default", "--yes"])
            assert r2.exit_code == 0, r2.output
            assert uninstalls == ["superpowers"]
```
