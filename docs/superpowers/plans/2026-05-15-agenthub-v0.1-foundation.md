# agenthub v0.1 Foundation Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a working v0.1.0 of agenthub — Python CLI that manages skills/commands/agents across Claude Code/Codex/OpenCode via per-tool profiles, with project-level overrides, multi-machine sync, and bootstrap onto a fresh machine.

**Architecture:** Python click CLI consuming a content hub repo. State engine renders dotfiles with `${VAR}` substitution and creates symlinks via a phased apply pipeline (parse → plan → pre-flight → stage → atomic-commit → cleanup). Per-tool targets implement a shared Protocol. Project bindings live in `hub/projects/<id>/config.toml` with per-machine cwd mappings in `~/.agenthub/state.json`.

**Tech Stack:** Python 3.11+, `click` (CLI), `tomlkit` (roundtrip TOML), `ruamel.yaml` (roundtrip YAML for frontmatter), `pytest` + `pytest-mock`, `ruff`, `pyright`. Distributed as a `pipx`-installable package.

**Scope of v0.1:** Pillars 1+2 (content sync + profile orchestration), Project bindings, `adopt`, Sync, Doctor/GC, Bootstrap. **Out of scope** (Plan 2 will cover): Pillar 3 (Source Ingestion), Pillar 4 (Plugin Management), Pillar 5 (MCP Server Management).

**Spec reference:** `docs/superpowers/specs/2026-05-15-alba-agent-tool-hub-design.md`

```python
from __future__ import annotations

from dataclasses import dataclass
from typing import Any

from agenthub import API_VERSION

KNOWN_KINDS = frozenset(
    {
        "Profile",
        "ProfileFragment",
        "ProjectConfig",
        "SourceIntent",
        "SourceLock",
        "PluginRef",
        "McpServer",
        "Config",
        "State",
        "HubVersion",
    }
)


class EnvelopeError(ValueError):
    """Raised when apiVersion/kind envelope is missing or wrong."""


@dataclass(frozen=True, slots=True)
class Envelope:
    api_version: str
    kind: str


def parse_envelope(doc: dict[str, Any], *, expected_kind: str) -> Envelope:
    api = doc.get("apiVersion")
    kind = doc.get("kind")
    if api is None or kind is None:
        raise EnvelopeError("missing apiVersion/kind envelope")
    if api != API_VERSION:
        raise EnvelopeError(f"unsupported apiVersion: {api} (this agenthub speaks {API_VERSION})")
    if kind not in KNOWN_KINDS:
        raise EnvelopeError(f"unknown kind: {kind}")
    if kind != expected_kind:
        raise EnvelopeError(f"expected kind {expected_kind!r} but got {kind!r}")
    return Envelope(api_version=api, kind=kind)
```

```python
from __future__ import annotations

from dataclasses import dataclass
from pathlib import Path

import tomlkit

from agenthub import API_VERSION
from agenthub.core.envelope import EnvelopeError, parse_envelope


class ConfigError(Exception):
    """Raised on config load/save errors."""


@dataclass(frozen=True, slots=True)
class Config:
    hub_path: Path
    # target_name -> (cli_path, enabled)
    targets: dict[str, tuple[str, bool]]
    secrets_env_path: Path


def load_config(path: Path) -> Config:
    if not path.exists():
        raise ConfigError(f"config not found at {path}")
    try:
        doc = tomlkit.parse(path.read_text())
    except tomlkit.exceptions.TOMLKitError as e:
        raise ConfigError(f"invalid TOML in {path}: {e}") from e
    try:
        parse_envelope(doc, expected_kind="Config")
    except EnvelopeError as e:
        raise ConfigError(str(e)) from e
    hub = doc.get("hub_path")
    if hub is None:
        raise ConfigError("missing hub_path")
    targets: dict[str, tuple[str, bool]] = {}
    targets_section = doc.get("targets", {})
    for name, body in targets_section.items():
        cli_path = body.get("cli_path", name)
        enabled = bool(body.get("enabled", True))
        targets[name] = (cli_path, enabled)
    render = doc.get("render", {})
    secrets = render.get("secrets_env_path", "~/.agenthub/secrets.env")
    return Config(
        hub_path=Path(hub).expanduser(),
        targets=targets,
        secrets_env_path=Path(secrets).expanduser(),
    )


def save_config(path: Path, cfg: Config) -> None:
    doc = tomlkit.document()
    doc["apiVersion"] = API_VERSION
    doc["kind"] = "Config"
    doc["hub_path"] = str(cfg.hub_path)
    targets_tbl = tomlkit.table()
    for name, (cli, enabled) in cfg.targets.items():
        inner = tomlkit.table()
        inner["enabled"] = enabled
        inner["cli_path"] = cli
        targets_tbl[name] = inner
    doc["targets"] = targets_tbl
    render = tomlkit.table()
    render["secrets_env_path"] = str(cfg.secrets_env_path)
    doc["render"] = render
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(tomlkit.dumps(doc))
```

```python
from __future__ import annotations

import json
from dataclasses import dataclass, field, replace
from pathlib import Path

from agenthub import API_VERSION


@dataclass(frozen=True, slots=True)
class State:
    active_profiles: dict[str, str] = field(default_factory=dict)
    project_mapping: dict[str, str] = field(default_factory=dict)

    def with_active(self, target: str, profile: str) -> State:
        new = dict(self.active_profiles)
        new[target] = profile
        return replace(self, active_profiles=new)

    def with_mapping(self, cwd: str, project_id: str) -> State:
        new = dict(self.project_mapping)
        new[cwd] = project_id
        return replace(self, project_mapping=new)


def load_state(path: Path) -> State:
    if not path.exists():
        return State()
    raw = json.loads(path.read_text())
    if raw.get("apiVersion") != API_VERSION:
        raise ValueError(f"unsupported state apiVersion: {raw.get('apiVersion')}")
    if raw.get("kind") != "State":
        raise ValueError(f"expected kind=State, got {raw.get('kind')}")
    spec = raw.get("spec", {})
    return State(
        active_profiles=dict(spec.get("active_profiles", {})),
        project_mapping=dict(spec.get("project_mapping", {})),
    )


def save_state(path: Path, state: State) -> None:
    payload = {
        "apiVersion": API_VERSION,
        "kind": "State",
        "spec": {
            "active_profiles": state.active_profiles,
            "project_mapping": state.project_mapping,
        },
    }
    path.parent.mkdir(parents=True, exist_ok=True)
    tmp = path.with_suffix(path.suffix + ".tmp")
    tmp.write_text(json.dumps(payload, indent=2, sort_keys=True))
    tmp.replace(path)
```

```python
from __future__ import annotations

from pathlib import Path


class SecretsError(Exception):
    """Raised on malformed secrets.env."""


def load_secrets(path: Path) -> dict[str, str]:
    if not path.exists():
        return {}
    out: dict[str, str] = {}
    for lineno, raw in enumerate(path.read_text().splitlines(), start=1):
        line = raw.strip()
        if not line or line.startswith("#"):
            continue
        if "=" not in line:
            raise SecretsError(f"line {lineno}: missing '=' in {raw!r}")
        key, _, value = line.partition("=")
        key = key.strip()
        value = value.strip()
        if (value.startswith('"') and value.endswith('"')) or (
            value.startswith("'") and value.endswith("'")
        ):
            value = value[1:-1]
        out[key] = value
    return out
```

```python
from __future__ import annotations

import re
from collections.abc import Mapping

# Order matters: handle $${VAR} escape before ${VAR}
_PLACEHOLDER = re.compile(r"\$\$\{([A-Za-z_][A-Za-z0-9_]*)\}|\$\{([A-Za-z_][A-Za-z0-9_]*)\}")


class RenderError(Exception):
    """Raised when a placeholder references an undefined variable."""


def render_string(text: str, env: Mapping[str, str]) -> str:
    missing: list[str] = []

    def repl(m: re.Match[str]) -> str:
        escaped, name = m.group(1), m.group(2)
        if escaped is not None:
            return "${" + escaped + "}"
        assert name is not None
        if name not in env:
            missing.append(name)
            return ""
        return env[name]

    rendered = _PLACEHOLDER.sub(repl, text)
    if missing:
        names = ", ".join(sorted(set(missing)))
        raise RenderError(f"undefined placeholder variable(s): {names}")
    return rendered


def render_bytes(data: bytes, env: Mapping[str, str]) -> bytes:
```

```python
from __future__ import annotations

from pathlib import Path

import tomlkit

from agenthub.core.envelope import EnvelopeError, parse_envelope

SUPPORTED_SCHEMA_VERSIONS: frozenset[int] = frozenset({1})


class HubError(Exception):
    """Raised when hub repo discovery/validation fails."""


def load_hub_version(hub_path: Path) -> int:
    version_file = hub_path / ".agenthub" / "version.toml"
    if not version_file.exists():
        raise HubError(f"{hub_path} is not a hub repo (no .agenthub/version.toml)")
    try:
        doc = tomlkit.parse(version_file.read_text())
    except tomlkit.exceptions.TOMLKitError as e:
        raise HubError(f"invalid hub version TOML: {e}") from e
    try:
        parse_envelope(doc, expected_kind="HubVersion")
    except EnvelopeError as e:
        raise HubError(str(e)) from e
    sv = doc.get("schema_version")
    if not isinstance(sv, int):
        raise HubError("schema_version must be an integer")
    if sv not in SUPPORTED_SCHEMA_VERSIONS:
        raise HubError(
            f"hub schema_version={sv} not supported by this agenthub "
            f"(supports {sorted(SUPPORTED_SCHEMA_VERSIONS)})"
        )
    return sv
```

```python
from __future__ import annotations

import io
from pathlib import Path
from typing import Any

from ruamel.yaml import YAML

_yaml = YAML(typ="rt")
_yaml.preserve_quotes = True


_DELIM = "---\n"


def read_frontmatter(path: Path) -> tuple[dict[str, Any], str]:
    text = path.read_text()
    if not text.startswith(_DELIM):
        return {}, text
    rest = text[len(_DELIM):]
    end = rest.find("\n" + _DELIM)
    if end == -1:
        return {}, text
    yaml_block = rest[:end + 1]
    body = rest[end + 1 + len(_DELIM):]
    data = _yaml.load(yaml_block) or {}
    return dict(data), body


def write_frontmatter(path: Path, frontmatter: dict[str, Any], body: str) -> None:
    buf = io.StringIO()
    _yaml.dump(frontmatter, buf)
    yaml_text = buf.getvalue()
    path.write_text(_DELIM + yaml_text + _DELIM + body)
```

```python
from __future__ import annotations

from dataclasses import dataclass
from enum import Enum
from pathlib import Path

from agenthub.core.frontmatter import read_frontmatter


class ContentKind(str, Enum):
    SKILL = "skill"
    COMMAND = "command"
    AGENT = "agent"


@dataclass(frozen=True, slots=True)
class Content:
    kind: ContentKind
    name: str
    path: Path                  # path to the SKILL.md (skill) or .md file (command/agent)
    tags: frozenset[str]


def _extract_tags(frontmatter: dict) -> frozenset[str]:
    meta = frontmatter.get("metadata") or {}
    tags = meta.get("tags") or []
    return frozenset(str(t).lower() for t in tags)


def _scan_skills(hub: Path) -> list[Content]:
    out: list[Content] = []
    skills_dir = hub / "skills"
    if not skills_dir.exists():
        return out
    for entry in sorted(skills_dir.iterdir()):
        if not entry.is_dir() or entry.name.startswith("."):
            continue
        skill_md = entry / "SKILL.md"
        if not skill_md.exists():
            continue
        fm, _ = read_frontmatter(skill_md)
        name = fm.get("name") or entry.name
        out.append(Content(ContentKind.SKILL, str(name), skill_md, _extract_tags(fm)))
    return out


def _scan_flat(hub: Path, dirname: str, kind: ContentKind) -> list[Content]:
    out: list[Content] = []
    d = hub / dirname
    if not d.exists():
        return out
    for f in sorted(d.iterdir()):
        if f.suffix != ".md" or f.name.startswith("."):
            continue
        if f.name.endswith(".agenthub-source.toml"):
            continue
        fm, _ = read_frontmatter(f)
        name = fm.get("name") or f.stem
        out.append(Content(kind, str(name), f, _extract_tags(fm)))
    return out


def scan_content(hub: Path) -> list[Content]:
    return (
        _scan_skills(hub)
        + _scan_flat(hub, "commands", ContentKind.COMMAND)
        + _scan_flat(hub, "agents", ContentKind.AGENT)
    )
```

```python
from __future__ import annotations

import logging
from collections.abc import Iterable, Mapping

logger = logging.getLogger(__name__)


class IncludeError(Exception):
    """Raised on invalid include entry syntax."""


def resolve_includes(entries: Iterable[str], tag_index: Mapping[str, Iterable[str]]) -> set[str]:
    out: set[str] = set()
    for raw in entries:
        e = raw.strip()
        if not e:
            continue
        if e.startswith("@"):
            tag = e[1:]
            if not tag:
                raise IncludeError(f"invalid tag reference: {raw!r}")
            members = list(tag_index.get(tag, ()))
            if not members:
                logger.warning("tag '@%s' resolves to empty set", tag)
            out.update(members)
        else:
            out.add(e)
    return out
```

```python
from __future__ import annotations

from dataclasses import dataclass, field
from pathlib import Path

import tomlkit

from agenthub.core.envelope import EnvelopeError, parse_envelope


class ProfileError(Exception):
    """Raised on missing/invalid profile."""


@dataclass(frozen=True, slots=True)
class Profile:
    target: str
    name: str
    profile_dir: Path
    skills: list[str] = field(default_factory=list)
    commands: list[str] = field(default_factory=list)
    agents: list[str] = field(default_factory=list)
    plugins: list[str] = field(default_factory=list)
    mcp: list[str] = field(default_factory=list)


_SECTIONS = ("skills", "commands", "agents", "plugins", "mcp")


def _parse_includes(doc: dict, expected_kind: str) -> dict[str, list[str]]:
    try:
        parse_envelope(doc, expected_kind=expected_kind)
    except EnvelopeError as e:
        raise ProfileError(str(e)) from e
    out: dict[str, list[str]] = {s: [] for s in _SECTIONS}
    for s in _SECTIONS:
        section = doc.get(s, {})
        out[s] = list(section.get("include", []))
    return out


def load_profile(hub: Path, target: str, name: str) -> Profile:
    profile_dir = hub / "profiles" / target / name
    if not profile_dir.exists():
        raise ProfileError(f"profile not found: {profile_dir}")
    main_file = profile_dir / "profile.toml"
    if not main_file.exists():
        raise ProfileError(f"profile.toml missing in {profile_dir}")
    main_doc = tomlkit.parse(main_file.read_text())
    merged = _parse_includes(main_doc, "Profile")
    fragments_dir = profile_dir / "includes.d"
    if fragments_dir.exists():
        for frag in sorted(fragments_dir.glob("*.toml")):
            frag_doc = tomlkit.parse(frag.read_text())
            frag_includes = _parse_includes(frag_doc, "ProfileFragment")
            for s in _SECTIONS:
                merged[s].extend(frag_includes[s])
    return Profile(
        target=target,
        name=name,
        profile_dir=profile_dir,
        skills=merged["skills"],
        commands=merged["commands"],
        agents=merged["agents"],
        plugins=merged["plugins"],
        mcp=merged["mcp"],
    )


def list_profiles(hub: Path, target: str) -> list[str]:
    base = hub / "profiles" / target
    if not base.exists():
        return []
    return sorted(d.name for d in base.iterdir() if d.is_dir() and (d / "profile.toml").exists())
```

```python
from importlib.metadata import entry_points

from agenthub.targets.base import Target


def load_targets() -> list[Target]:
    out: list[Target] = []
    for ep in entry_points(group="agenthub.targets"):
        cls = ep.load()
        instance = cls()
        if not isinstance(instance, Target):
            raise RuntimeError(f"entry-point {ep.name!r} does not satisfy Target protocol")
        out.append(instance)
    return out


def load_target(name: str) -> Target:
    for t in load_targets():
        if t.name == name:
            return t
    raise KeyError(f"no target registered with name {name!r}")
```

```python
from __future__ import annotations

from dataclasses import dataclass
from enum import Enum
from pathlib import Path


class PlanOp(str, Enum):
    LINK = "link"
    UNLINK = "unlink"
    WRITE = "write"


@dataclass(frozen=True, slots=True)
class PlanItem:
    op: PlanOp
    kind: str          # "skill" | "command" | "agent" | "dotfile"
    name: str
    target: Path       # where the link/file lands
    source: Path | None  # link target for LINK; None for UNLINK; rendered tmp for WRITE


def _current_link_target(p: Path) -> Path | None:
    if not p.is_symlink():
        return None
    try:
        return p.resolve(strict=False)
    except OSError:
        return None


def compute_link_plan(
    *,
    desired: dict[str, Path],
    live_root: Path,
    kind: str = "skill",
) -> list[PlanItem]:
    """Diff desired symlinks against live_root.

    desired:    name -> hub source dir/file
    live_root:  e.g. ~/.claude/skills (must exist; may be empty)
    """
    items: list[PlanItem] = []
    live_names: set[str] = set()
    if live_root.exists():
        for entry in live_root.iterdir():
            live_names.add(entry.name)
    desired_names = set(desired)
    for name in sorted(desired_names | live_names):
        target = live_root / name
        if name in desired_names and name in live_names:
            current = _current_link_target(target)
            if current is not None and current == desired[name].resolve():
                continue
            items.append(PlanItem(PlanOp.UNLINK, kind, name, target, None))
            items.append(PlanItem(PlanOp.LINK, kind, name, target, desired[name]))
        elif name in desired_names:
            items.append(PlanItem(PlanOp.LINK, kind, name, target, desired[name]))
        else:
            items.append(PlanItem(PlanOp.UNLINK, kind, name, target, None))
    return items
```

```python
from __future__ import annotations

import os
from collections.abc import Iterable
from pathlib import Path

from agenthub.core.plan import PlanItem, PlanOp


class ApplyError(Exception):
    """Raised when apply detects an unsafe condition (pre-flight failure, etc.)."""


def _preflight(plan: Iterable[PlanItem]) -> None:
    for item in plan:
        if item.op is not PlanOp.LINK:
            continue
        t = item.target
        if t.exists() and not t.is_symlink():
            raise ApplyError(f"unmanaged file at {t}; refusing to overwrite")
        if t.is_symlink():
            # We will replace this symlink later; safe.
            continue


def execute_plan(
    plan: list[PlanItem],
    *,
    dry_run: bool,
    staging_root: Path,
) -> None:
    if not plan:
        return
    _preflight(plan)
    if dry_run:
        return
    staging_root.mkdir(parents=True, exist_ok=True)
    for item in plan:
        if item.op is PlanOp.UNLINK:
            if item.target.is_symlink() or item.target.exists():
                item.target.unlink(missing_ok=True)
        elif item.op is PlanOp.LINK:
            assert item.source is not None
            item.target.parent.mkdir(parents=True, exist_ok=True)
            if item.target.is_symlink() or item.target.exists():
                item.target.unlink()
            os.symlink(item.source, item.target)
        elif item.op is PlanOp.WRITE:
            assert item.source is not None
            item.target.parent.mkdir(parents=True, exist_ok=True)
            tmp = item.target.with_suffix(item.target.suffix + ".tmp")
            tmp.write_bytes(item.source.read_bytes())
            tmp.replace(item.target)
```

```python
from __future__ import annotations

from collections.abc import Mapping
from pathlib import Path

from agenthub.core.apply import execute_plan
from agenthub.core.content import ContentKind, scan_content
from agenthub.core.include import resolve_includes
from agenthub.core.plan import PlanItem, PlanOp, compute_link_plan
from agenthub.core.profile import load_profile
from agenthub.core.tag import build_tag_index
from agenthub.targets.base import Target


def _resolve_content(
    hub: Path, kind: ContentKind, includes: list[str]
) -> dict[str, Path]:
    items = [c for c in scan_content(hub) if c.kind is kind]
    index = build_tag_index(items)
    names = resolve_includes(includes, index)
    by_name = {c.name: c for c in items}
    desired: dict[str, Path] = {}
    for n in names:
        c = by_name.get(n)
        if c is None:
            continue
        desired[n] = c.path if kind is not ContentKind.SKILL else c.path.parent
    return desired


def build_plan(
    *,
    hub: Path,
    target: Target,
    profile_name: str,
    env: Mapping[str, str],
    project_root: Path | None = None,
) -> list[PlanItem]:
    """Construct the full apply plan for one (target, profile) pair."""
    profile = load_profile(hub, target.name, profile_name)
    target_root = target.project_root(project_root) if project_root else target.global_root()
    plan: list[PlanItem] = []
    for kind, includes in (
        (ContentKind.SKILL, profile.skills),
        (ContentKind.COMMAND, profile.commands),
        (ContentKind.AGENT, profile.agents),
    ):
        desired = _resolve_content(hub, kind, includes)
        subdir = target_root / target.content_subdir(kind.value)
        plan.extend(compute_link_plan(desired=desired, live_root=subdir, kind=kind.value))
    return plan


def apply_global(
    hub: Path,
    target: Target,
    profile_name: str,
    env: Mapping[str, str],
    staging_root: Path,
    *,
    dry_run: bool,
) -> list[PlanItem]:
    plan = build_plan(hub=hub, target=target, profile_name=profile_name, env=env)
    execute_plan(plan, dry_run=dry_run, staging_root=staging_root)
    return plan


def apply_project(
    hub: Path,
    target: Target,
    project_root: Path,
    includes: dict[str, list[str]],
    env: Mapping[str, str],
    staging_root: Path,
    *,
    dry_run: bool,
) -> list[PlanItem]:
    """Apply a per-target project config (already parsed) under <project>/.<tool>/."""
    target_root = target.project_root(project_root)
    plan: list[PlanItem] = []
    for kind_name in ("skill", "command", "agent"):
        kind = ContentKind(kind_name)
        desired = _resolve_content(hub, kind, includes.get(f"{kind_name}s", []))
        subdir = target_root / target.content_subdir(kind_name)
        plan.extend(compute_link_plan(desired=desired, live_root=subdir, kind=kind_name))
    execute_plan(plan, dry_run=dry_run, staging_root=staging_root)
    return plan
```

```python
from __future__ import annotations

import json
from dataclasses import dataclass, field
from pathlib import Path

from agenthub import API_VERSION


@dataclass(frozen=True, slots=True)
class LastApply:
    target: str
    profile: str
    scope: str                    # "global" | "project"
    links: dict[str, str] = field(default_factory=dict)  # rel_target_path -> hub_source_abs


def read_manifest(path: Path) -> LastApply | None:
    if not path.exists():
        return None
    raw = json.loads(path.read_text())
    return LastApply(
        target=raw["target"],
        profile=raw["profile"],
        scope=raw["scope"],
        links=dict(raw.get("links", {})),
    )


def write_manifest(path: Path, m: LastApply) -> None:
    payload = {
        "apiVersion": API_VERSION,
        "kind": "LastApply",
        "spec": {"target": m.target, "profile": m.profile, "scope": m.scope, "links": m.links},
    }
    path.parent.mkdir(parents=True, exist_ok=True)
    tmp = path.with_suffix(path.suffix + ".tmp")
    tmp.write_text(json.dumps(payload, indent=2, sort_keys=True))
    tmp.replace(path)
```

```python
from __future__ import annotations

import sys
from pathlib import Path

import click

from agenthub.cli._ctx import (
    default_config_path,
    default_manifest_path,
    default_staging_root,
    default_state_path,
    resolve_hub,
)
from agenthub.core.config import load_config
from agenthub.core.orchestrator import apply_global
from agenthub.core.plan import PlanOp
from agenthub.core.secrets import load_secrets
from agenthub.core.state import load_state, save_state
from agenthub.targets.registry import load_target, load_targets


@click.command("apply")
@click.option("--target", "target_name", help="claude|codex|opencode|all (default: all)")
@click.option("--profile", "profile_name", help="Profile name; default = sticky from state.json or 'default'.")
@click.pass_context
def apply_cmd(ctx: click.Context, target_name: str | None, profile_name: str | None) -> None:
    """Apply the active profile to the given target."""
    hub_override = ctx.obj.get("hub_override")
    dry_run: bool = ctx.obj.get("dry_run", False)
    yes: bool = ctx.obj.get("yes", False)
    verbose: bool = ctx.obj.get("verbose", False)

    res = resolve_hub(flag_value=hub_override, config_path=default_config_path())
    cfg = load_config(default_config_path())
    state = load_state(default_state_path())
    env = load_secrets(cfg.secrets_env_path)

    targets = (
        load_targets() if target_name in (None, "all") else [load_target(target_name)]
    )

    for target in targets:
        if not target.detect():
            click.echo(f"– {target.name}: not installed (skipped)")
            continue
        active = profile_name or state.active_profiles.get(target.name, "default")
        click.echo(f"== {target.name}: applying profile '{active}'")
        plan = apply_global(
            hub=res.hub_path,
            target=target,
            profile_name=active,
            env=env,
            staging_root=default_staging_root(),
            manifest_path=default_manifest_path(),
            dry_run=True,  # dry to print plan first
        )
        if not plan:
            click.echo("  (nothing to do)")
            continue
        for item in plan:
            if verbose:
                click.echo(f"  {item.op.value:6s} {item.target}")
            else:
                sigil = "+" if item.op is PlanOp.LINK else "-"
                click.echo(f"  {sigil} {item.kind} {item.name}")
        if dry_run:
            continue
        if not yes and not click.confirm("proceed?", default=True):
            click.echo("aborted.")
            sys.exit(1)
        apply_global(
            hub=res.hub_path,
            target=target,
            profile_name=active,
            env=env,
            staging_root=default_staging_root(),
            manifest_path=default_manifest_path(),
            dry_run=False,
        )
        state = state.with_active(target.name, active)
        save_state(default_state_path(), state)
        click.echo(f"✓ {target.name}: profile '{active}' applied")
```

```python
from __future__ import annotations

import datetime
from dataclasses import dataclass, field
from pathlib import Path

import tomlkit

from agenthub import API_VERSION
from agenthub.core.envelope import EnvelopeError, parse_envelope


class ProjectError(Exception):
    """Raised on missing/invalid project config."""


@dataclass(frozen=True, slots=True)
class ProjectConfig:
    meta: dict[str, str] = field(default_factory=dict)
    # target -> {section -> include list}; section in {"skills","commands","agents","plugins","mcp"}
    includes: dict[str, dict[str, list[str]]] = field(default_factory=dict)

    def includes_for(self, target: str) -> dict[str, list[str]]:
        return dict(self.includes.get(target, {}))


_TARGETS = ("claude", "codex", "opencode")
_SECTIONS = ("skills", "commands", "agents", "plugins", "mcp")


def load_project_config(path: Path) -> ProjectConfig:
    if not path.exists():
        raise ProjectError(f"project config not found: {path}")
    doc = tomlkit.parse(path.read_text())
    try:
        parse_envelope(doc, expected_kind="ProjectConfig")
    except EnvelopeError as e:
        raise ProjectError(str(e)) from e
    meta = dict(doc.get("meta", {}))
    includes: dict[str, dict[str, list[str]]] = {}
    for t in _TARGETS:
        target_section = doc.get(t, {})
        sub: dict[str, list[str]] = {}
        for s in _SECTIONS:
            sec = target_section.get(s, {})
            inc = sec.get("include", [])
            if inc:
                sub[s] = list(inc)
        if sub:
            includes[t] = sub
    return ProjectConfig(meta=meta, includes=includes)


def save_project_config(path: Path, cfg: ProjectConfig) -> None:
    doc = tomlkit.document()
    doc["apiVersion"] = API_VERSION
    doc["kind"] = "ProjectConfig"
    meta = dict(cfg.meta)
    if "created_at" not in meta:
        meta["created_at"] = datetime.date.today().isoformat()
    meta_tbl = tomlkit.table()
    for k, v in meta.items():
        meta_tbl[k] = v
    doc["meta"] = meta_tbl
    for t in _TARGETS:
        t_includes = cfg.includes.get(t, {})
        if not t_includes:
            continue
        t_tbl = tomlkit.table()
        for s in _SECTIONS:
            inc = t_includes.get(s)
            if not inc:
                continue
            inner = tomlkit.table()
            inner["include"] = inc
            t_tbl[s] = inner
        doc[t] = t_tbl
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(tomlkit.dumps(doc))
```

```python
from __future__ import annotations

import subprocess
from pathlib import Path

import click

from agenthub.cli._ctx import default_config_path, resolve_hub


def _git(args: list[str], cwd: Path) -> tuple[int, str, str]:
    p = subprocess.run(["git", *args], cwd=cwd, capture_output=True, text=True)
    return p.returncode, p.stdout, p.stderr


@click.command("sync")
@click.pass_context
def sync_cmd(ctx: click.Context) -> None:
    """git pull + push the hub repo."""
    res = resolve_hub(flag_value=ctx.obj.get("hub_override"), config_path=default_config_path())
    hub = res.hub_path
    dry: bool = ctx.obj.get("dry_run", False)
    # Check working tree clean
    rc, out, _ = _git(["status", "--porcelain"], hub)
    if rc != 0:
        raise click.ClickException("not a git repo")
    if out.strip():
        raise click.ClickException("hub has uncommitted changes; commit or stash first")
    # Check remote
    rc, out, _ = _git(["remote"], hub)
    if rc != 0 or not out.strip():
        raise click.ClickException("hub has no remote configured")
    if dry:
        click.echo("would: git pull --ff-only")
        click.echo("would: git push")
        return
    rc, out, err = _git(["pull", "--ff-only"], hub)
    if rc != 0:
        raise click.ClickException(f"git pull failed: {err.strip()}")
    click.echo(out.rstrip())
    rc, out, err = _git(["push"], hub)
    if rc != 0:
        raise click.ClickException(f"git push failed: {err.strip()}")
    click.echo("✓ sync complete")
```

```python
from __future__ import annotations

import re
import shutil
from pathlib import Path

import click

from agenthub.cli._ctx import default_config_path, default_state_path, resolve_hub
from agenthub.core.config import load_config
from agenthub.core.secrets import load_secrets
from agenthub.core.state import load_state

_PLACEHOLDER_RE = re.compile(r"\$\{([A-Za-z_][A-Za-z0-9_]*)\}")


def _scan_placeholders(profile_root: Path) -> set[str]:
    out: set[str] = set()
    if not profile_root.exists():
        return out
    for f in profile_root.rglob("*"):
        if f.is_file() and not f.name.startswith("."):
            try:
                txt = f.read_text(errors="ignore")
            except OSError:
                continue
            for m in _PLACEHOLDER_RE.finditer(txt):
                out.add(m.group(1))
    return out


@click.command("doctor")
@click.pass_context
def doctor_cmd(ctx: click.Context) -> None:
    """Report stale state, missing secrets, orphan links."""
    res = resolve_hub(flag_value=ctx.obj.get("hub_override"), config_path=default_config_path())
    hub = res.hub_path
    state = load_state(default_state_path())
    cfg = load_config(default_config_path())
    issues = 0

    # Stale project mappings
    for cwd, pid in state.project_mapping.items():
        hub_entry = hub / "projects" / pid / "config.toml"
        if not hub_entry.exists():
            click.echo(f"⚠ stale project mapping: cwd={cwd} → '{pid}' (hub entry missing)")
            issues += 1
        if not Path(cwd).exists():
            click.echo(f"⚠ mapped cwd no longer exists: {cwd}")
            issues += 1

    # Missing secrets
    referenced = _scan_placeholders(hub / "profiles")
    referenced |= _scan_placeholders(hub / "mcp-servers")
    available = set(load_secrets(cfg.secrets_env_path).keys())
    missing = referenced - available
    if missing:
        click.echo(f"⚠ {len(missing)} placeholder vars not in secrets.env: {sorted(missing)}")
        issues += 1

    if issues == 0:
        click.echo("✓ no issues")


@click.command("gc")
@click.pass_context
def gc_cmd(ctx: click.Context) -> None:
    """Clean up agenthub temp/staging dirs."""
    staging = Path.home() / ".agenthub" / "staging"
    if staging.exists():
        shutil.rmtree(staging)
        click.echo(f"✓ removed {staging}")
    else:
        click.echo("(nothing to gc)")
```

```python
def _render_dotfiles(profile_dir: Path, target_root: Path, env: Mapping[str, str], staging_root: Path) -> list[PlanItem]:
    from agenthub.core.render import render_string
    items: list[PlanItem] = []
    if not profile_dir.exists():
        return items
    for f in sorted(profile_dir.iterdir()):
        if not f.is_file():
            continue
        # skip profile.toml + includes.d
        if f.name == "profile.toml":
            continue
        try:
            text = f.read_text()
        except UnicodeDecodeError:
            text = None
        if text is not None:
            rendered = render_string(text, env)
        else:
            rendered = f.read_bytes().decode("latin1", errors="ignore")
        rendered_path = staging_root / f.name
        rendered_path.parent.mkdir(parents=True, exist_ok=True)
        rendered_path.write_text(rendered)
        items.append(
            PlanItem(op=PlanOp.WRITE, kind="dotfile", name=f.name,
                     target=target_root / f.name, source=rendered_path)
        )
    return items
```
