# agenthub v0.1.1 Hotfix Implementation Plan

> **For agentic workers:** Implement task-by-task. Each step's checkbox is for tracking.

**Goal:** Fix critical apply-pipeline bugs found in v0.1.0 smoke testing on a non-empty target directory. Specifically: stop agenthub from deleting unmanaged content, add proper pre-flight safety, and filter junk files. Manifest-driven UNLINK is the core architectural fix.

**Architecture:** Change the "what to manage" model from "everything in live_root" to "only entries this agenthub installation previously created (per last-apply manifest)". The first apply on a populated dir will then leave user content alone; subsequent applies only remove what they previously added.

**Tech Stack:** Same as v0.1.0 — Python 3.11+, click, tomlkit, pytest.

**Reference spec:** `docs/superpowers/specs/2026-05-15-alba-agent-tool-hub-design.md` §10.1 (Pre-flight collision check), §8.2 (doctor + manifest semantics).

**Bugs being fixed (from smoke test report):**
1. CRITICAL — `compute_link_plan` does set-diff between desired and live, erroneously UNLINKs unmanaged content
2. HIGH — `_preflight` only checks LINK ops; UNLINK on real dirs crashes apply mid-flight
3. MEDIUM — `.DS_Store` and other non-symlink junk treated as artifacts
4. MEDIUM — No use of last-apply manifest to inform "what's mine"
5. LOW — Emoji output inconsistency between CLI commands

**Out of scope (deferred):**
- True staging + atomic rename of the entire tool dir (spec §10.1 second bullet). The fix below keeps per-op execution, but with the manifest-driven model it's safe — only agenthub's own creations can be partially affected on crash, never user data.

Replace the entire function with:

```python
def compute_link_plan(
    *,
    desired: dict[str, Path],
    live_root: Path,
    previous_managed: frozenset[str] = frozenset(),
    kind: str = "skill",
) -> list[PlanItem]:
    """Diff desired symlinks against the set of entries we previously created.

    Args:
        desired: name -> hub source path (what the active profile wants linked here)
        live_root: e.g. ~/.claude/skills (the target dir; need not exist yet)
        previous_managed: names this agenthub installation previously created here,
            from last-apply.json. Empty set means "first apply" — we will not UNLINK
            anything (user's pre-existing content is preserved).
        kind: artifact kind for PlanItem.kind

    Behavior:
        - For each name in desired: emit LINK (always; executor checks if already correct).
        - For each name in (previous_managed - desired): emit UNLINK (we own it; remove).
        - Entries in live_root that are neither desired nor in previous_managed are IGNORED.
    """
    items: list[PlanItem] = []
    desired_names = set(desired)
    to_unlink = previous_managed - desired_names
    for name in sorted(desired_names):
        target = live_root / name
        current = _current_link_target(target)
        if current is not None and current == desired[name].resolve():
            continue  # symlink already correct; no-op
        items.append(PlanItem(PlanOp.LINK, kind, name, target, desired[name]))
    for name in sorted(to_unlink):
        items.append(PlanItem(PlanOp.UNLINK, kind, name, live_root / name, None))
    return items
```

Notes:
- `live_root` is no longer iterated for the diff; we don't care about strangers.
- The function no longer needs to filter dotfiles — they were only a problem because we iterated live_root.
- Existing tests in test_plan.py that don't pass `previous_managed` will keep working (defaults to empty frozenset) but their semantics change. Update them in Step 2.

```python
def _preflight(plan: Iterable[PlanItem]) -> None:
    for item in plan:
        t = item.target
        if item.op is PlanOp.LINK:
            # LINK: target must not be a non-symlink real file/dir.
            # A symlink (any) is OK — we'll replace it. Missing is OK — we'll create.
            if t.exists() and not t.is_symlink():
                raise ApplyError(
                    f"refusing to overwrite unmanaged file at {t}; "
                    f"move or remove it manually if you want agenthub to manage that name"
                )
        elif item.op is PlanOp.UNLINK:
            # UNLINK: target must be a symlink (i.e. created by us).
            # If it's a real file/dir, the manifest is stale or external code changed
            # things; refuse rather than risk data loss.
            if not t.is_symlink():
                raise ApplyError(
                    f"refusing to unlink {t}: not a symlink "
                    f"(manifest may be stale; run `agenthub doctor` to inspect)"
                )
        # WRITE: handled separately; target overwrite via tmp + rename is safe.
```

def previous_managed_for(
    manifest: LastApply | None,
    *,
    live_root: Path,
) -> frozenset[str]:
    """Extract the basenames of entries created under live_root in the last apply.

    Used to feed compute_link_plan so it only proposes UNLINK for entries we actually
    created. Returns empty frozenset if no manifest exists (first apply).
    """
    if manifest is None:
        return frozenset()
    out: set[str] = set()
    live_str = str(live_root.resolve())
    for target_str in manifest.links:
        p = Path(target_str)
        # Treat as ours if its parent (resolved) matches live_root (resolved).
        try:
            parent = p.parent.resolve()
        except OSError:
            continue
        if str(parent) == live_str:
            out.add(p.name)
    return frozenset(out)
```

def apply_global(
    hub: Path,
    target: Target,
    profile_name: str,
    env: Mapping[str, str],
    staging_root: Path,
    manifest_path: Path,
    *,
    dry_run: bool,
) -> list[PlanItem]:
    prior = read_manifest(manifest_path)
    plan = build_plan(
        hub=hub, target=target, profile_name=profile_name, env=env, manifest=prior,
    )
    execute_plan(plan, dry_run=dry_run, staging_root=staging_root)
    if not dry_run:
        links = {}
        for item in plan:
            if item.op is PlanOp.LINK and item.source is not None:
                links[str(item.target)] = str(item.source.resolve())
        write_manifest(
            manifest_path,
            LastApply(target=target.name, profile=profile_name, scope="global", links=links),
        )
    return plan
```

def apply_project(
    hub: Path,
    target: Target,
    project_root: Path,
    includes: dict[str, list[str]],
    env: Mapping[str, str],
    staging_root: Path,
    *,
    dry_run: bool,
    manifest_path: Path | None = None,
) -> list[PlanItem]:
    target_root = target.project_root(project_root)
    prior = read_manifest(manifest_path) if manifest_path else None
    plan: list[PlanItem] = []
    for kind_name in ("skill", "command", "agent"):
        kind = ContentKind(kind_name)
        desired = _resolve_content(hub, kind, includes.get(f"{kind_name}s", []))

            desired=desired, live_root=subdir,
            previous_managed=previous_managed, kind=kind_name,
        ))
    execute_plan(plan, dry_run=dry_run, staging_root=staging_root)
    return plan

    result = runner.invoke(main, ["apply", "--target", "claude", "--profile", "default", "--yes"])
