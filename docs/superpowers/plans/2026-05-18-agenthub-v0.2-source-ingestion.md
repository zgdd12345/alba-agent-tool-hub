# agenthub v0.2 — Pillar 3 Source Ingestion Implementation Plan

> **For agentic workers:** Use superpowers:subagent-driven-development to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build Pillar 3 (Source Ingestion) on top of v0.1.2. Enable importing skills/commands/agents from upstream git repos with sidecar + lock metadata, autostash+rebase update flow, and a 5-choice conflict resolver. After v0.2.0, Alba can `agenthub source import --url <git-url> --path skills/foo` to pull from anthropic/skills or any other repo.

**Architecture:** Each imported artifact gets a paired `.agenthub-source.toml` (intent: url/path/ref/kind) and `.agenthub-source.lock.toml` (resolved: sha/commit_title/imported_at). Imports use a shallow-clone cache at `~/.agenthub/content-cache/<url-hash>/`. Updates compare upstream HEAD vs lock.sha; on uncommitted local divergence, run autostash → apply upstream → replay; only on replay failure, prompt the user with 5 options.

**Tech stack:** Same as v0.1.x. Adds `subprocess` calls to `git`. No new pip dependencies.

**Reference spec:** `docs/superpowers/specs/2026-05-15-alba-agent-tool-hub-design.md` §4 (Source Ingestion).

**Scope of v0.2:**
- Sources: `git`, `git-subdir` (full repo or single subdir, the two most-used kinds)
- CLI: `source import`, `source update`, `source ls`, `source attach`, `source detach`
- Update conflict UX: autostash+rebase default → 5-choice fallback (stash/abort/diff/merge/detach)

**Out of scope (Plan 3+ or later):**
- Sources `github-release`, `http`, `npm` (rarely used; can be added incrementally)
- `agenthub source refresh` (re-resolve sha for all, without changing intent)
- Bulk import (`--from-manifest`)

```python
from __future__ import annotations

from dataclasses import dataclass
from datetime import datetime
from enum import StrEnum
from pathlib import Path

import tomlkit

from agenthub import API_VERSION
from agenthub.core.envelope import EnvelopeError, parse_envelope


class SourceError(Exception):
    """Raised on sidecar/lock load/save errors."""


class SourceKind(StrEnum):
    GIT = "git"
    GIT_SUBDIR = "git-subdir"
    # github | github_release | http | npm — added later


@dataclass(frozen=True, slots=True)
class SourceIntent:
    kind: SourceKind
    url: str
    path: str = ""               # subdir within the repo; empty = whole repo is one artifact
    ref: str = "main"            # branch / tag / commit-ref
    ref_type: str = "branch"     # "branch" | "tag" | "commit"


@dataclass(frozen=True, slots=True)
class SourceLock:
    sha: str                     # resolved commit SHA
    commit_title: str
    imported_at: datetime
    last_updated: datetime


def load_intent(path: Path) -> SourceIntent:
    if not path.exists():
        raise SourceError(f"intent not found: {path}")
    doc = tomlkit.parse(path.read_text())
    try:
        parse_envelope(doc, expected_kind="SourceIntent")
    except EnvelopeError as e:
        raise SourceError(str(e)) from e
    source = doc.get("source", {})
    kind_str = source.get("type")
    if kind_str is None:
        raise SourceError("missing [source].type")
    try:
        kind = SourceKind(kind_str)
    except ValueError as e:
        raise SourceError(f"unknown source type: {kind_str!r}") from e
    return SourceIntent(
        kind=kind,
        url=source["url"],
        path=source.get("path", ""),
        ref=source.get("ref", "main"),
        ref_type=source.get("ref_type", "branch"),
    )


def save_intent(path: Path, intent: SourceIntent) -> None:
    doc = tomlkit.document()
    doc["apiVersion"] = API_VERSION
    doc["kind"] = "SourceIntent"
    src = tomlkit.table()
    src["type"] = intent.kind.value
    src["url"] = intent.url
    if intent.path:
        src["path"] = intent.path
    src["ref"] = intent.ref
    src["ref_type"] = intent.ref_type
    doc["source"] = src
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(tomlkit.dumps(doc))


def load_lock(path: Path) -> SourceLock:
    if not path.exists():
        raise SourceError(f"lock not found: {path}")
    doc = tomlkit.parse(path.read_text())
    try:
        parse_envelope(doc, expected_kind="SourceLock")
    except EnvelopeError as e:
        raise SourceError(str(e)) from e
    return SourceLock(
        sha=doc["sha"],
        commit_title=doc.get("commit_title", ""),
        imported_at=datetime.fromisoformat(doc["imported_at"]),
        last_updated=datetime.fromisoformat(doc["last_updated"]),
    )


def save_lock(path: Path, lock: SourceLock) -> None:
    doc = tomlkit.document()
    doc["apiVersion"] = API_VERSION
    doc["kind"] = "SourceLock"
    doc["sha"] = lock.sha
    doc["commit_title"] = lock.commit_title
    doc["imported_at"] = lock.imported_at.isoformat()
    doc["last_updated"] = lock.last_updated.isoformat()
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(tomlkit.dumps(doc))
```

```python
from __future__ import annotations

import hashlib
from pathlib import Path


def cache_path_for(url: str, cache_root: Path) -> Path:
    """Return a stable per-URL cache directory path (no I/O)."""
    digest = hashlib.sha256(url.encode("utf-8")).hexdigest()[:16]
    return cache_root / digest


def ensure_cache_root(cache_root: Path) -> Path:
    cache_root.mkdir(parents=True, exist_ok=True)
    return cache_root


def default_cache_root() -> Path:
    return Path.home() / ".agenthub" / "content-cache"
```

```python
import pytest

from agenthub.core.git_source import GitFetchError, fetch_to_cache, resolve_sha


def _run(cmd: list[str], cwd: Path) -> None:
    subprocess.check_call(cmd, cwd=cwd)


def _init_repo_with_content(p: Path, content: dict[str, str]) -> str:
    """Init a bare-ish git repo with file contents; return HEAD SHA."""
    p.mkdir(parents=True, exist_ok=True)
    _run(["git", "init", "-q", "-b", "main"], p)
    _run(["git", "config", "user.email", "test@example.com"], p)
    _run(["git", "config", "user.name", "Test"], p)
    for relpath, body in content.items():
        full = p / relpath
        full.parent.mkdir(parents=True, exist_ok=True)
        full.write_text(body)
    _run(["git", "add", "-A"], p)
    _run(["git", "commit", "-q", "-m", "initial"], p)
    sha = subprocess.check_output(["git", "rev-parse", "HEAD"], cwd=p, text=True).strip()
    return sha


def test_fetch_to_cache_clones_and_checks_out(tmp_path: Path):
    upstream = tmp_path / "upstream"
    expected_sha = _init_repo_with_content(upstream, {"skills/foo/SKILL.md": "# foo"})
    cache_root = tmp_path / "cache"
    cache = fetch_to_cache(url=str(upstream), ref="main", cache_root=cache_root)

    assert resolve_sha(cache) == expected_sha


def test_fetch_rejects_invalid_url(tmp_path: Path):
    with pytest.raises(GitFetchError):
        fetch_to_cache(url="/no/such/repo", ref="main", cache_root=tmp_path / "cache")


def test_fetch_reuses_cache_on_second_call(tmp_path: Path):
    upstream = tmp_path / "upstream"
    _init_repo_with_content(upstream, {"x.txt": "v1"})
    cache_root = tmp_path / "cache"
    first = fetch_to_cache(url=str(upstream), ref="main", cache_root=cache_root)
    # Mutate upstream
    (upstream / "x.txt").write_text("v2")
    _run(["git", "add", "-A"], upstream)
    _run(["git", "commit", "-q", "-m", "update"], upstream)
```

```python
from __future__ import annotations

import subprocess
from pathlib import Path

from agenthub.core.cache import cache_path_for, ensure_cache_root


class GitFetchError(Exception):
    """Raised when a git fetch / clone / checkout fails."""


def _git(args: list[str], cwd: Path | None = None) -> tuple[int, str, str]:
    p = subprocess.run(
        ["git", *args],
        cwd=cwd,
        capture_output=True,
        text=True,
        check=False,
    )
    return p.returncode, p.stdout, p.stderr


def fetch_to_cache(*, url: str, ref: str, cache_root: Path) -> Path:
    """Clone-or-update url at ref into the cache. Returns the cache dir.

    Idempotent: if the cache dir exists, fetch + checkout the requested ref.
    """
    ensure_cache_root(cache_root)
    dest = cache_path_for(url, cache_root)
    if not (dest / ".git").exists():
        # Fresh shallow clone
        rc, _, err = _git(["clone", "--depth=1", "--no-tags", "-b", ref, url, str(dest)])
        if rc != 0:
            # Fallback: clone without -b (some refs aren't branches), then checkout
            if dest.exists():
                _remove_dir(dest)
            rc2, _, err2 = _git(["clone", "--depth=1", "--no-tags", url, str(dest)])
            if rc2 != 0:
                raise GitFetchError(f"git clone failed: {err.strip() or err2.strip()}")
            rc3, _, err3 = _git(["fetch", "--depth=1", "origin", ref], cwd=dest)
            if rc3 != 0:
                raise GitFetchError(f"git fetch failed: {err3.strip()}")
            rc4, _, err4 = _git(["checkout", ref], cwd=dest)
            if rc4 != 0:
                raise GitFetchError(f"git checkout failed: {err4.strip()}")
        return dest
    # Existing cache: fetch + checkout
    rc, _, err = _git(["fetch", "--depth=1", "origin", ref], cwd=dest)
    if rc != 0:
        raise GitFetchError(f"git fetch failed: {err.strip()}")
    rc, _, err = _git(["checkout", ref], cwd=dest)
    if rc != 0:
        # try detached checkout to FETCH_HEAD
        rc2, _, err2 = _git(["checkout", "FETCH_HEAD"], cwd=dest)
        if rc2 != 0:
            raise GitFetchError(f"git checkout failed: {err.strip() or err2.strip()}")
    return dest


def resolve_sha(cache_dir: Path) -> str:
    rc, out, err = _git(["rev-parse", "HEAD"], cwd=cache_dir)
    if rc != 0:
        raise GitFetchError(f"git rev-parse failed: {err.strip()}")
    return out.strip()


def resolve_commit_title(cache_dir: Path, sha: str) -> str:
    rc, out, _ = _git(["log", "-1", "--format=%s", sha], cwd=cache_dir)
    return out.strip() if rc == 0 else ""


def _remove_dir(p: Path) -> None:
    import shutil
    shutil.rmtree(p)
```

```python
from __future__ import annotations

import shutil
from dataclasses import dataclass
from datetime import datetime, timezone
from pathlib import Path

from agenthub.core.cache import default_cache_root
from agenthub.core.git_source import fetch_to_cache, resolve_commit_title, resolve_sha
from agenthub.core.source import (
    SourceIntent,
    SourceKind,
    SourceLock,
    save_intent,
    save_lock,
)


class IngestError(Exception):
    """Raised on ingest failures."""


@dataclass(frozen=True, slots=True)
class ImportResult:
    target_path: Path        # where in hub the artifact landed
    sha: str
    commit_title: str


def _hub_subdir_for_kind(kind: str) -> str:
    return {"skill": "skills", "command": "commands", "agent": "agents"}[kind]


def _is_dir_artifact(kind: str) -> bool:
    return kind == "skill"


def import_source(
    *,
    hub: Path,
    url: str,
    path: str,                  # subdir within upstream; "" = entire repo
    kind: str,                  # "skill" | "command" | "agent"
    name: str | None,           # local name in hub; default = basename(path) or repo name
    ref: str = "main",
    cache_root: Path | None = None,
) -> ImportResult:
    cache_root = cache_root or default_cache_root()
    # Resolve target name
    if name is None:
        name = Path(path).name if path else Path(url.rstrip("/")).stem
        # strip .git suffix from repo name
        if name.endswith(".git"):
            name = name[: -len(".git")]
    # Determine hub destination
    subdir = _hub_subdir_for_kind(kind)
    if _is_dir_artifact(kind):
        dest = hub / subdir / name
        if dest.exists():
            raise IngestError(f"already exists in hub: {dest}")
    else:
        dest_file = hub / subdir / f"{name}.md"
        if dest_file.exists():
            raise IngestError(f"already exists in hub: {dest_file}")
        dest = dest_file.parent
    # Fetch upstream
    cache = fetch_to_cache(url=url, ref=ref, cache_root=cache_root)
    sha = resolve_sha(cache)
    title = resolve_commit_title(cache, sha)
    src_root = cache / path if path else cache
    if not src_root.exists():
        raise IngestError(f"upstream path not found: {path}")
    # Copy into hub
    if _is_dir_artifact(kind):
        shutil.copytree(src_root, dest, ignore=shutil.ignore_patterns(".git"))
        sidecar_dir = dest
    else:
        # commands / agents: a single .md file expected at src_root or src_root.md
        candidate = src_root if src_root.is_file() else src_root.with_suffix(".md")
        if not candidate.is_file():
            raise IngestError(f"expected {kind} .md file at upstream path: {path}")
        target_md = hub / subdir / f"{name}.md"
        target_md.parent.mkdir(parents=True, exist_ok=True)
        shutil.copy2(candidate, target_md)
        sidecar_dir = target_md.parent
    # Write sidecar + lock
    intent = SourceIntent(
        kind=SourceKind.GIT_SUBDIR if path else SourceKind.GIT,
        url=url,
        path=path,
        ref=ref,
        ref_type="branch",  # we accept tag refs too but default annotation is branch
    )
    now = datetime.now(timezone.utc)
    lock = SourceLock(sha=sha, commit_title=title, imported_at=now, last_updated=now)
    if _is_dir_artifact(kind):
        save_intent(dest / ".agenthub-source.toml", intent)
        save_lock(dest / ".agenthub-source.lock.toml", lock)
        result_path = dest
    else:
        save_intent(hub / subdir / f"{name}.agenthub-source.toml", intent)
        save_lock(hub / subdir / f"{name}.agenthub-source.lock.toml", lock)
        result_path = hub / subdir / f"{name}.md"
    return ImportResult(target_path=result_path, sha=sha, commit_title=title)
```

```python
class UpdateResult:
    target_path: Path
    sha: str                    # new sha (== old sha if up_to_date)
    up_to_date: bool            # True if no changes pulled
    conflict: bool = False      # True if local edits prevented clean update
    conflict_files: tuple[str, ...] = ()


def _sidecar_locations(hub: Path, name: str, kind: str) -> tuple[Path, Path, Path]:
    """Return (intent_path, lock_path, content_root) for the named artifact."""
    subdir = _hub_subdir_for_kind(kind)
    if _is_dir_artifact(kind):
        content_root = hub / subdir / name
        return (
            content_root / ".agenthub-source.toml",
            content_root / ".agenthub-source.lock.toml",
            content_root,
        )
    return (
        hub / subdir / f"{name}.agenthub-source.toml",
        hub / subdir / f"{name}.agenthub-source.lock.toml",
        hub / subdir / f"{name}.md",
    )


def update_source(
    *,
    hub: Path,
    name: str,
    kind: str,
    ref: str | None = None,
    cache_root: Path | None = None,
) -> UpdateResult:
    cache_root = cache_root or default_cache_root()
    intent_path, lock_path, content_root = _sidecar_locations(hub, name, kind)
    intent = load_intent(intent_path)
    lock = load_lock(lock_path)
    target_ref = ref or intent.ref
    cache = fetch_to_cache(url=intent.url, ref=target_ref, cache_root=cache_root)
    new_sha = resolve_sha(cache)
    if new_sha == lock.sha:
        return UpdateResult(target_path=content_root, sha=new_sha, up_to_date=True)
    src_root = cache / intent.path if intent.path else cache
    # For v0.2 simplification: replace content wholesale.
    # Conflict detection: if hub content differs from old upstream snapshot, it means
    # user edited locally. We stash and warn (autostash+replay loop is deferred until S6).
    title = resolve_commit_title(cache, new_sha)
    now = datetime.now(timezone.utc)
    if _is_dir_artifact(kind):
        # Diff via git: clone old + new shas, compare against hub. For v0.2 we accept
        # "wholesale overwrite if no diff between hub and old-snapshot"; richer diff
        # detection lands in S6.
        if _hub_diverges_from_lock(content_root, intent, lock, cache_root):
            return UpdateResult(
                target_path=content_root,
                sha=lock.sha,
                up_to_date=False,
                conflict=True,
                conflict_files=("(local edits detected; run `agenthub source update --resolve` to handle)",),
            )
        shutil.rmtree(content_root)
        shutil.copytree(src_root, content_root, ignore=shutil.ignore_patterns(".git"))
        save_intent(intent_path, intent)
        save_lock(lock_path, SourceLock(sha=new_sha, commit_title=title,
                                        imported_at=lock.imported_at, last_updated=now))
    else:
        if _hub_diverges_from_lock(content_root, intent, lock, cache_root):
            return UpdateResult(
                target_path=content_root,
                sha=lock.sha,
                up_to_date=False,
                conflict=True,
                conflict_files=("(local edits detected)",),
            )
        candidate = src_root if src_root.is_file() else src_root.with_suffix(".md")
        shutil.copy2(candidate, content_root)
        save_intent(intent_path, intent)
        save_lock(lock_path, SourceLock(sha=new_sha, commit_title=title,
                                        imported_at=lock.imported_at, last_updated=now))
```

```python
def _hub_diverges_from_lock(
    content_root: Path,
    intent: SourceIntent,
    lock: SourceLock,
    cache_root: Path,
) -> bool:
    """Compare hub content against the upstream snapshot at lock.sha.

    Returns True if hub content has been edited locally since import.
    """
    # Fetch the locked sha into a dedicated cache slot
    snapshot_cache = cache_path_for(intent.url + "#" + lock.sha, cache_root)
    if not (snapshot_cache / ".git").exists():
        snapshot_cache.parent.mkdir(parents=True, exist_ok=True)
        rc, _, err = _git(["clone", "--no-tags", intent.url, str(snapshot_cache)])
        if rc != 0:
            # Can't fetch the snapshot — be safe, declare conflict
            return True
        rc, _, _ = _git(["checkout", lock.sha], cwd=snapshot_cache)
        if rc != 0:
            return True
    snapshot_root = (snapshot_cache / intent.path) if intent.path else snapshot_cache
    if not snapshot_root.exists():
        return True
    return _trees_differ(content_root, snapshot_root)


def _trees_differ(a: Path, b: Path) -> bool:
    """Lightweight recursive comparison. Returns True if differ.

    Skips `.git` dirs and `.agenthub-source*.toml` sidecars (they exist only in `a`).
    """
    a_files = _list_files(a, skip={".git"}, skip_suffixes={".agenthub-source.toml", ".agenthub-source.lock.toml"})
    b_files = _list_files(b, skip={".git"}, skip_suffixes={".agenthub-source.toml", ".agenthub-source.lock.toml"})
    if a_files.keys() != b_files.keys():
        return True
    for rel, sz_a in a_files.items():
        sz_b = b_files[rel]
        if sz_a != sz_b:
            return True
        if (a / rel).read_bytes() != (b / rel).read_bytes():
            return True
    return False


def _list_files(root: Path, *, skip: set[str], skip_suffixes: set[str]) -> dict[str, int]:
    out: dict[str, int] = {}
    if root.is_file():
        out[root.name] = root.stat().st_size
        return out
    for f in root.rglob("*"):
        if not f.is_file():
            continue
        rel_parts = f.relative_to(root).parts
        if any(part in skip for part in rel_parts):
            continue
        if any(f.name.endswith(s) for s in skip_suffixes):
            continue
        out[str(f.relative_to(root))] = f.stat().st_size
    return out
```

```python
from __future__ import annotations

from datetime import datetime, timezone
from pathlib import Path

import click

from agenthub.cli._ctx import default_config_path, resolve_hub
from agenthub.core.cache import default_cache_root
from agenthub.core.ingest import IngestError, import_source, update_source
from agenthub.core.source import (

@click.group("source")
def source_grp() -> None:
    """Manage upstream-sourced content."""


def _hub_from_ctx(ctx: click.Context) -> Path:
    return resolve_hub(
        flag_value=ctx.obj.get("hub_override"),
        config_path=default_config_path(),
    ).hub_path


@source_grp.command("import")
@click.option("--url", required=True)
@click.option("--path", "subpath", default="")
@click.option("--kind", type=click.Choice(["skill", "command", "agent"]), default="skill")
@click.option("--name", default=None)
@click.option("--ref", default="main")
@click.pass_context
def import_cmd(ctx: click.Context, url: str, subpath: str, kind: str,
               name: str | None, ref: str) -> None:

    try:
        result = import_source(
            hub=hub, url=url, path=subpath, kind=kind, name=name, ref=ref,
            cache_root=default_cache_root(),
        )
    except IngestError as e:
        raise click.ClickException(str(e))
    click.echo(f"imported into {result.target_path} (sha={result.sha[:12]})")
    click.echo("not in any profile/project yet; add to enable.")


@source_grp.command("ls")
@click.pass_context
def ls_cmd(ctx: click.Context) -> None:
    hub = _hub_from_ctx(ctx)
    rows = []
    for subdir, kind in (("skills", "skill"), ("commands", "command"), ("agents", "agent")):
        base = hub / subdir
        if not base.exists():
            continue
        if kind == "skill":
            for d in base.iterdir():
                if not d.is_dir():
                    continue
                sidecar = d / ".agenthub-source.toml"
                if sidecar.exists():
                    rows.append((kind, d.name, sidecar, d / ".agenthub-source.lock.toml"))
        else:
            for sidecar in base.glob("*.agenthub-source.toml"):
                name = sidecar.name[: -len(".agenthub-source.toml")]
                rows.append((kind, name, sidecar, base / f"{name}.agenthub-source.lock.toml"))
    if not rows:
        click.echo("(no upstream-tracked content)")
        return
    for kind, name, sidecar, lock_p in rows:
        intent = load_intent(sidecar)
        sha_short = ""
        if lock_p.exists():
            lock = load_lock(lock_p)
            sha_short = lock.sha[:12]
        click.echo(f"{kind:8s} {name:24s} {intent.url}#{intent.ref}  sha={sha_short}")


@source_grp.command("detach")
@click.argument("name")
@click.option("--kind", type=click.Choice(["skill", "command", "agent"]), default="skill")
@click.option("--yes", "-y", is_flag=True)
@click.pass_context
def detach_cmd(ctx: click.Context, name: str, kind: str, yes: bool) -> None:
    hub = _hub_from_ctx(ctx)
    from agenthub.core.ingest import _sidecar_locations
    intent_path, lock_path, _ = _sidecar_locations(hub, name, kind)
    if not intent_path.exists():
        raise click.ClickException(f"no sidecar at {intent_path}")
    if not yes and not click.confirm(f"detach {kind} {name} from upstream tracking?", default=True):
        return
    intent_path.unlink(missing_ok=True)
    lock_path.unlink(missing_ok=True)
    click.echo(f"detached {kind} {name}")


@source_grp.command("attach")
@click.argument("name")
@click.option("--kind", type=click.Choice(["skill", "command", "agent"]), default="skill")
@click.option("--url", required=True)
@click.option("--path", "subpath", default="")
@click.option("--ref", default="main")
@click.option("--sha", required=True)
@click.pass_context
def attach_cmd(ctx: click.Context, name: str, kind: str, url: str, subpath: str,
               ref: str, sha: str) -> None:
    """Annotate an existing hub artifact with upstream tracking (without re-fetching)."""
    hub = _hub_from_ctx(ctx)
    from agenthub.core.ingest import _sidecar_locations
    intent_path, lock_path, content_root = _sidecar_locations(hub, name, kind)
    if not content_root.exists():
        raise click.ClickException(f"no {kind} at {content_root}")
    intent = SourceIntent(
        kind=SourceKind.GIT_SUBDIR if subpath else SourceKind.GIT,
        url=url, path=subpath, ref=ref, ref_type="branch",
    )
    now = datetime.now(timezone.utc)
    lock = SourceLock(sha=sha, commit_title="", imported_at=now, last_updated=now)
    save_intent(intent_path, intent)
    save_lock(lock_path, lock)
    click.echo(f"attached {kind} {name} → {url}@{sha[:12]}")


@source_grp.command("update")
@click.argument("name", required=False)
@click.option("--kind", type=click.Choice(["skill", "command", "agent"]), default="skill")
@click.option("--ref", default=None)
@click.option("--all", "update_all", is_flag=True)
@click.pass_context
def update_cmd(ctx: click.Context, name: str | None, kind: str, ref: str | None,
               update_all: bool) -> None:
    hub = _hub_from_ctx(ctx)
    if update_all:
        # Iterate over all sidecars
        targets = []
        for subdir, k in (("skills", "skill"), ("commands", "command"), ("agents", "agent")):
            base = hub / subdir
            if not base.exists():
                continue
            if k == "skill":
                for d in base.iterdir():
                    if d.is_dir() and (d / ".agenthub-source.toml").exists():
                        targets.append((d.name, k))
            else:
                for sidecar in base.glob("*.agenthub-source.toml"):
                    targets.append((sidecar.name[: -len(".agenthub-source.toml")], k))
        if not targets:
            click.echo("(nothing to update)")
            return
        for n, k in targets:
            _do_update(hub, n, k, ref)
    else:
        if name is None:
            raise click.ClickException("provide <name> or use --all")
        _do_update(hub, name, kind, ref)


def _do_update(hub: Path, name: str, kind: str, ref: str | None) -> None:
    try:
        result = update_source(hub=hub, name=name, kind=kind, ref=ref,
                               cache_root=default_cache_root())
    except IngestError as e:
        raise click.ClickException(str(e))
    if result.conflict:
        click.echo(f"CONFLICT: {kind} {name} has local edits; skipping. "
                   f"Detach or manually resolve, then re-run.")
        return
    if result.up_to_date:
        click.echo(f"OK: {kind} {name} already up to date (sha={result.sha[:12]})")
    else:
        click.echo(f"updated: {kind} {name} → sha={result.sha[:12]}")
```
