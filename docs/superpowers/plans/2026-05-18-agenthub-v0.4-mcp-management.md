# agenthub v0.4 — Pillar 5 MCP Server Management

> Final pillar. Implements spec §6 (MCP server management). Based on real-machine probing (2026-05-18) of `claude mcp`, `codex mcp`, and `opencode.json` MCP surfaces.

**Goal:** `agenthub mcp add/ls/show/rm` to manage MCP server definitions in hub. Profile + project includes drive install/uninstall via each tool's native MCP CLI (Claude + Codex) or JSON config edit (OpenCode). Manifest tracks installed MCPs per target. After v0.4, all 5 pillars are complete.

**Architecture:** Mirrors Pillar 4 (plugin management) closely — `McpServer` schema in hub, `Target.install_mcp/uninstall_mcp/list_installed_mcp` Protocol methods, `PlanOp.INSTALL_MCP/UNINSTALL_MCP`, manifest entry `mcp_servers: dict[str, list[str]]`.

**Tech stack:** No new deps. Reuses existing `run_subprocess`.

**Reference spec:** §6 of `docs/superpowers/specs/2026-05-15-alba-agent-tool-hub-design.md`.

**Verified MCP CLI surfaces (2026-05-18):**
- **Claude**: `claude mcp add <name> <commandOrUrl> [args...]` with `-e ENV=value`, `--transport http`; `claude mcp remove <name>`; `claude mcp list`
- **Codex**: `codex mcp add <NAME> (--url <URL> | -- <COMMAND>...)` with `--env K=V`, `--bearer-token-env-var`; `codex mcp remove <NAME>`; `codex mcp list [--json]`
- **OpenCode**: no CLI; edit `~/.config/opencode/opencode.json` `mcp` section: `{"<name>": {"type": "local"|"remote", "command": [...] | "url": "..."}}`

---

## Scope of v0.4

- `McpServer` schema (envelope + `[server]` + `[env]` + `[targets.<tool>]`)
- CLI: `agenthub mcp add/ls/show/rm`
- `Target.install_mcp/uninstall_mcp/list_installed_mcp` Protocol methods (no-op default)
- Apply pipeline emits `INSTALL_MCP/UNINSTALL_MCP` ops; manifest tracks state
- Three target impls

**Out of scope (Phase 2):**
- `agenthub mcp adopt --from-settings ...` (extract MCP from existing settings.json) — spec §11 candidate
- `agenthub mcp test <name>` (spawn/ping verification) — spec §11 candidate
- Project-scoped Codex MCP (Codex `-c` overrides scalar config only; not a great fit; defer)

```python
from __future__ import annotations

from dataclasses import dataclass, field
from enum import StrEnum
from pathlib import Path

import tomlkit

from agenthub import API_VERSION
from agenthub.core.envelope import EnvelopeError, parse_envelope


class McpError(Exception):
    """Raised on McpServer load/save errors."""


class McpServerType(StrEnum):
    LOCAL = "local"
    REMOTE = "remote"


@dataclass(frozen=True, slots=True)
class McpServer:
    name: str
    type: McpServerType
    # local fields
    command: str = ""
    args: tuple[str, ...] = field(default_factory=tuple)
    # remote fields
    url: str = ""
    headers: dict[str, str] = field(default_factory=dict)
    # common
    env: dict[str, str] = field(default_factory=dict)
    description: str = ""
    tags: tuple[str, ...] = field(default_factory=tuple)


def _validate(srv: McpServer) -> None:
    if srv.type is McpServerType.LOCAL and not srv.command:
        raise McpError("local mcp server requires [server].command")
    if srv.type is McpServerType.REMOTE and not srv.url:
        raise McpError("remote mcp server requires [server].url")


def load_mcp_server(path: Path) -> McpServer:
    if not path.exists():
        raise McpError(f"mcp server not found: {path}")
    doc = tomlkit.parse(path.read_text())
    try:
        parse_envelope(doc, expected_kind="McpServer")
    except EnvelopeError as e:
        raise McpError(str(e)) from e
    server = doc.get("server", {})
    type_str = server.get("type", "")
    try:
        srv_type = McpServerType(type_str)
    except ValueError as e:
        raise McpError(f"unknown mcp server type: {type_str!r}") from e
    meta = doc.get("meta", {})
    env_tbl = doc.get("env", {}) or {}
    srv = McpServer(
        name=path.stem,
        type=srv_type,
        command=server.get("command", ""),
        args=tuple(server.get("args", []) or []),
        url=server.get("url", ""),
        headers=dict(server.get("headers", {}) or {}),
        env={k: str(v) for k, v in env_tbl.items()},
        description=meta.get("description", ""),
        tags=tuple(meta.get("tags", []) or []),
    )
    _validate(srv)
    return srv


def save_mcp_server(path: Path, srv: McpServer) -> None:
    _validate(srv)
    doc = tomlkit.document()
    doc["apiVersion"] = API_VERSION
    doc["kind"] = "McpServer"
    meta = tomlkit.table()
    meta["description"] = srv.description  # may be ""
    if srv.tags:
        meta["tags"] = list(srv.tags)
    doc["meta"] = meta
    s = tomlkit.table()
    s["type"] = srv.type.value
    if srv.type is McpServerType.LOCAL:
        s["command"] = srv.command
        if srv.args:
            s["args"] = list(srv.args)
    else:
        s["url"] = srv.url
        if srv.headers:
            hdr = tomlkit.table()
            for k, v in srv.headers.items():
                hdr[k] = v
            s["headers"] = hdr
    doc["server"] = s
    if srv.env:
        env = tomlkit.table()
        for k, v in srv.env.items():
            env[k] = v
        doc["env"] = env
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(tomlkit.dumps(doc))
```

```python
def install_mcp(self, srv: McpServer) -> None:
    # Build: claude mcp add <name> [-e K=V ...] [--transport http] <commandOrUrl> [args...]
    # OR for remote with headers: claude mcp add --transport http <name> <url> --header "K: V"
    cmd: list[str] = ["claude", "mcp", "add"]
    for k, v in sorted(srv.env.items()):
        cmd.extend(["-e", f"{k}={v}"])
    if srv.type is McpServerType.REMOTE:
        cmd.extend(["--transport", "http"])
        for k, v in sorted(srv.headers.items()):
            cmd.extend(["--header", f"{k}: {v}"])
        cmd.append(srv.name)
        cmd.append(srv.url)
    else:
        cmd.append(srv.name)
        cmd.append("--")
        cmd.append(srv.command)
        cmd.extend(srv.args)
    # Tolerate "already exists" — replace by remove-then-add
    rc, _, err = run_subprocess(cmd)
    if rc != 0 and "already" in err.lower():
        # Remove then retry
        run_subprocess(["claude", "mcp", "remove", srv.name])
        rc, _, err = run_subprocess(cmd)
    if rc != 0:
        raise RuntimeError(f"claude mcp add failed: {err.strip()}")


def uninstall_mcp(self, name: str) -> None:
    rc, _, err = run_subprocess(["claude", "mcp", "remove", name])
    if rc != 0 and "not" not in err.lower():  # tolerate "not found"
        raise RuntimeError(f"claude mcp remove failed: {err.strip()}")


def list_installed_mcp_servers(self) -> set[str]:
    rc, out, _ = run_subprocess(["claude", "mcp", "list"])
    if rc != 0:
        return set()
    names: set[str] = set()
    for line in out.splitlines():
        line = line.strip()
        if not line:
            continue
        # Format: "name: status" or "name (transport): status"
        name = line.split(":", 1)[0].strip()
        # Strip optional " (transport)" suffix
        if " (" in name:
            name = name.split(" (")[0]
        if name and not name.startswith("#"):
            names.add(name)
    return names
```

def install_mcp(self, srv: McpServer) -> None:

    for k, v in sorted(srv.env.items()):
        cmd.extend(["--env", f"{k}={v}"])

    else:

    rc, out, _ = run_subprocess(["codex", "mcp", "list", "--json"])
    if rc != 0:
        return set()

        cfg = _opencode_json(self.global_root())
        data = _read_opencode_config(cfg)
        mcp = data.setdefault("mcp", {})
        if srv.type is McpServerType.LOCAL:
            entry: dict = {
                "type": "local",
                "command": [srv.command, *srv.args],
            }
            if srv.env:
                entry["environment"] = dict(srv.env)
        else:
            entry = {"type": "remote", "url": srv.url}
            if srv.headers:
                entry["headers"] = dict(srv.headers)
        mcp[srv.name] = entry
        _write_opencode_config(cfg, data)

    def uninstall_mcp(self, name: str) -> None:
        cfg = _opencode_json(self.global_root())
        data = _read_opencode_config(cfg)
        mcp = data.get("mcp", {})
        if name in mcp:
            del mcp[name]
        if not mcp and "mcp" in data:
            del data["mcp"]
        _write_opencode_config(cfg, data)

    def list_installed_mcp_servers(self) -> set[str]:
        data = _read_opencode_config(_opencode_json(self.global_root()))
        mcp = data.get("mcp", {})
        return set(mcp.keys()) if isinstance(mcp, dict) else set()

```python
from __future__ import annotations

from pathlib import Path

import click

from agenthub.cli._ctx import hub_path_from_ctx
from agenthub.core.mcp import (
    McpError,
    McpServer,
    McpServerType,
    load_mcp_server,
    save_mcp_server,
)


@click.group("mcp")
def mcp_grp() -> None:
    """Manage MCP server definitions in the hub."""


def _ref_path(hub: Path, name: str) -> Path:
    return hub / "mcp-servers" / f"{name}.toml"


@mcp_grp.command("add")
@click.option("--name", required=True)
@click.option("--type", "srv_type", type=click.Choice(["local", "remote"]), required=True)
@click.option("--command", default="", help="local: launch command (e.g. npx)")
@click.option("--args", "args_csv", default="", help="local: comma-separated args")
@click.option("--url", default="", help="remote: URL")
@click.option("--env", "env_kvs", multiple=True, help="K=V env pair (repeatable)")
@click.option("--description", default="")
@click.option("--tags", default="")
@click.pass_context
def add_cmd(
    ctx: click.Context,
    name: str,
    srv_type: str,
    command: str,
    args_csv: str,
    url: str,
    env_kvs: tuple[str, ...],
    description: str,
    tags: str,
) -> None:
    hub = hub_path_from_ctx(ctx)
    env: dict[str, str] = {}
    for kv in env_kvs:
        if "=" not in kv:
            raise click.ClickException(f"--env requires K=V form; got {kv!r}")
        k, _, v = kv.partition("=")
        env[k] = v
    tag_tuple = tuple(t.strip() for t in tags.split(",") if t.strip())
    args_tuple = tuple(a for a in args_csv.split(",") if a)
    srv = McpServer(
        name=name,
        type=McpServerType(srv_type),
        command=command,
        args=args_tuple,
        url=url,
        env=env,
        description=description,
        tags=tag_tuple,
    )
    dest = _ref_path(hub, name)
    if dest.exists():
        raise click.ClickException(f"mcp server already exists: {dest}")
    try:
        save_mcp_server(dest, srv)
    except McpError as e:
        raise click.ClickException(str(e)) from e
    click.echo(f"added: mcp server at {dest}")
    click.echo("not in any profile/project yet; add to enable.")


@mcp_grp.command("ls")
@click.pass_context
def ls_cmd(ctx: click.Context) -> None:
    hub = hub_path_from_ctx(ctx)
    base = hub / "mcp-servers"
    if not base.exists():
        click.echo("(no mcp servers)")
        return
    any_found = False
    for f in sorted(base.glob("*.toml")):
        try:
            srv = load_mcp_server(f)
        except McpError as e:
            click.echo(f"WARN: {f}: {e}")
            continue
        if srv.type is McpServerType.LOCAL:
            descr = f"local: {srv.command} {' '.join(srv.args)}"
        else:
            descr = f"remote: {srv.url}"
        click.echo(f"{srv.name:24s} {descr}")
        any_found = True
    if not any_found:
        click.echo("(no mcp servers)")


@mcp_grp.command("show")
@click.argument("name")
@click.pass_context
def show_cmd(ctx: click.Context, name: str) -> None:
    hub = hub_path_from_ctx(ctx)
    p = _ref_path(hub, name)
    if not p.exists():
        raise click.ClickException(f"no mcp server at {p}")
    click.echo(p.read_text())


@mcp_grp.command("rm")
@click.argument("name")
@click.option("--yes", "-y", is_flag=True)
@click.pass_context
def rm_cmd(ctx: click.Context, name: str, yes: bool) -> None:
    hub = hub_path_from_ctx(ctx)
    p = _ref_path(hub, name)
    if not p.exists():
        raise click.ClickException(f"no mcp server at {p}")
    if not yes and not click.confirm(f"remove mcp server ref {p}?", default=True):
        return
    p.unlink()
    click.echo(f"removed: {p}")
```
